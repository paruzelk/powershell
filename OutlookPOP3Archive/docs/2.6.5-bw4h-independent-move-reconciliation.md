    $outlook=[Runtime.InteropServices.Marshal]::GetActiveObject('Outlook.Application')
    $ns=$outlook.GetNamespace('MAPI')
    if([string]$ns.CurrentProfileName -cne 'Outlook'){throw 'Wrong profile.'}
    $srcStore=$null
    for($i=1;$i -le $ns.Stores.Count;$i++){
        $s=$ns.Stores.Item($i)
        try{
          if([string]::Equals([IO.Path]::GetFullPath([string]$s.FilePath),
              [IO.Path]::GetFullPath($source),
              [StringComparison]::OrdinalIgnoreCase)){$srcStore=$s}
        }catch{}
    }
    if($null -eq $srcStore){throw 'Source POP3 PST not mounted.'}
    $sourceFolderEntryId=[string]$srcStore.GetDefaultFolder(6).EntryID

    $auditScript=Join-Path (Get-Location).Path 'Core\Diagnostics\Test-OPA265BW4hIndependentReconciliationReadOnly.ps1'
